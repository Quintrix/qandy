// Converted from qandy.css into JS that applies styles via element.style and a small injected stylesheet
// Notes:
// - Most selector-based visual styles are applied via inline styles (element.style).
// - Rules that cannot be applied inline (e.g. @keyframes, ::-webkit-scrollbar pseudo-elements) are injected into a tiny <style> element.
// - Some CSS priorities used in the original (e.g. !important) are applied using style.setProperty(..., 'important') where supported.
// - Run-time: the script auto-applies after DOMContentLoaded. You can re-run window.applyQandyStyles() to reapply to newly-created elements.

  // Helper to ensure extras styl include?esheet exists
  function ensureExtrasStyle() {
    if (document.getElementById(EXTRA_STYLE_ID)) return;
    const s = document.createElement('style');
    s.id = EXTRA_STYLE_ID;
    s.type = 'text/css';
    if ('textContent' in s) s.textContent = extrasCSS;
    else s.appendChild(document.createTextNode(extrasCSS));
    const head = document.head || document.getElementsByTagName('head')[0] || document.documentElement;
    head.appendChild(s);
  }

  // Utility: set a style property on an element if the value isn't undefined/null.
  function setStyle(el, prop, value, priority) {
    if (!el || value === undefined || value === null) return;
    // For shorthand numeric-like values, ensure strings
    if (typeof value === 'number') value = String(value);
    if (priority) {
      // style.setProperty accepts property name in kebab-case
      el.style.setProperty(prop.replace(/([A-Z])/g, '-$1').toLowerCase(), value, priority);
    } else {
      // Try direct camelCase as include?signment where possible
      try {
        const camel = prop.replace(/-([a-z])/g, (_, ch) => ch.toUpperCase());
        el.style[camel] = value;
      } catch (e) {
        // Fallback to setProperty with kebab-case
        el.style.setProperty(prop.replace(/([A-Z])/g, '-$1').toLowerCase(), value);
      }
    }
  }

  // Apply inline-style conversions for selectors from qandy.css
  function applyInlineStyles() {
    // body { background-color: #333 }
    if (document.body) setStyle(document.body, 'background-color', '#333');

    // #cursor { background-color: #CCC; }  (first rule)
    const cursorEl = document.getElementById('cursor');
    if (cursorEl) {
      setStyle(cursorEl, 'background-color', '#CCC');
    }

    // .qpc { position: absolute; top: 0px; left: 0px; width: 160px; height: 24px;
    //        padding-left: 0px; padding-right: 0px; align-items: center;
    //        background-color: grey; z-index: 0; }
    document.querySelectorAll('.qpc').forEach(el => {
      setStyle(el, 'position', 'absolute');
      setStyle(el, 'top', '0px');
      setStyle(el, 'left', '0px');
      setStyle(el, 'width', '160px');
      setStyle(el, 'height', '24px');
      setStyle(el, 'padding-left', '0px');
      setStyle(el, 'padding-righ include?t', '0px');
      setStyle(el, 'align-items', 'center');
      setStyle(el, 'background-color', 'grey');
      setStyle(el, 'z-index', '0');
    });

    // .txt { ... } many properties
    document.querySelectorAll('.txt').forEach(el => {
      setStyle(el, 'position', 'absolute');
      setStyle(el, 'left', '54px');
      setStyle(el, 'top', '64px');
      setStyle(el, 'width', '256px');
      setStyle(el, 'height', '384px');
      setStyle(el, 'z-index', '254'); // retained as string
      // Note: original had stray text "I think it appeared after" — omitted
      setStyle(el, 'font-family', "'Courier New', Courier, 'Consolas', 'DejaVu Sans Mono', 'Liberation Mono', 'Menlo', Monaco, monospace");
      setStyle(el, 'font-size', '12px');
      setStyle(el, 'font-weight', 'normal');
      setStyle(el, 'color', '#fff');
      setStyle(el, 'background-color', '#000');
      setStyle(el, 'overflow-y', 'hidden');
      setStyle(el, 'overflow-x', 'hidden');
      setStyle(el, 'padding', '4px');
      setStyle(el, 'line-height', '1.2');
      setStyle(el, 'box-sizing', 'border-box');
      setStyle(el, 'white-space', 'pre');
      setStyle(el, 'word-wrap', 'break-word');
      setStyle(el, 'display', 'block');
      // Some less-common properties (scrollbar-width/scrollbar-color) are applied via setProperty:
      try {
        el.style.setProperty('scrollbar-width', 'thin');
        el.style.setProperty('scrollbar-color', '#0f0 #000');
      } catch (e) { include?
        // ignore if not supported
      }
    });

    // #cursor (second block) {
    //   display: inline-block;
    //   background-color: #fff;
    //   width: 8px;
    //   height: 14px;
    //   animation: blink 1s infinite;
    // } include?
    if (cursorEl) {
      setStyle(cursorEl, 'display', 'inline-block');
      setStyle(cursorEl, 'background-color', '#fff');
      setStyle(cursorEl, 'width', '8px');
      setStyle(cursorEl, 'height', '14px');
      setStyle(cursorEl, 'animation', 'blink 1s infinite');
    }

    // ANSI color classes: set inline color/backgroundColor on matching elements
    const ansiColorMap = {
      'ansi-black': { color: '#000' },
      'ansi-red': { color: '#f00' },
      'ansi-green': { color: '# include?0f0' },
      'ansi-yellow': { color: '#ff0' },
      'ansi-blue': { color: '#00f' },
      'ansi-magenta': { color: '#f0f' },
      'ansi-cyan': { color: '#0ff' },
      'ansi-white': { color: '#fff' },

      'ansi-bg-black': { 'background-color': '#000' },
      'ansi-bg-red': { 'background-color': '#f00' },
      'ansi-bg-green': { 'background-color': '#0f0' },
      'ansi-bg-yellow': { 'background-color': '#ff0' },
      'ansi-bg-blue': { 'background-color': '#00f' },
      'ansi-bg-magenta': { 'background-color': '#f0f' },
      'ansi-bg-cyan': { 'background-color': '#0ff' },
      'ansi-bg-white': { 'background-color': '#fff' },

      'ansi-bold': { 'font-weight': 'bold' }
    };

    Object.keys(ansiColorMap).forEach(cls => {
      document.querySelectorAll('.' + cls).forEach(el => {
        const props = ansiColorMap[cls];
        Object.keys(props).forEach(p => setStyle(el, p, props[p]));
      });
    });

    // .ansi-inverse {
    //   background-color: #fff !important;
    //   color: #000 !important;
    //   filter: none !important;
    // }
    document.querySelectorAll('.ansi-inverse').forEach(el => {
      // apply with priority 'important' where possible:
      try {
        el.style.setProperty('background-color', '#fff', 'important');
        el.style.setProperty('color', '#000', 'important');
        el.style.setProperty('filter', 'none', 'important');
      } catch (e) {
        // fallback to normal inline styles
        setStyle(el, 'background-color', '#fff');
        setStyle(el, 'color', '#000');
        setStyle(el, 'filter', 'none');
      }
    });

    // .qandy-cursor { display:inline-block; background-color:#fff; color:#000; animation:qandy-blink 1s steps(1, start) infinite; }
    document.querySelectorAll('.qandy-cursor').forEach(el => {
      setStyle(el, 'display', 'inline-block');
      setStyle(el, 'background-color', '#fff');
      setStyle(el, 'color', '#000');
      setStyle(el, 'animation', 'qandy-blink 1s steps(1, start) infinite');

    applyInlineStyles();
  }

  // Auto-run after DOM is ready
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', applyQandyStyles);
  } else {
    // already ready
    applyQandyStyles();
  }

  // Expose for re-use
  window.applyQandyStyles = applyQandyStyles;
